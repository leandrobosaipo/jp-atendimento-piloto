var stateReadCache = null;
function stateFileIdentity(stat){return [stat.dev,stat.ino,stat.size,stat.mtimeNs,stat.ctimeNs].join(':');}
function load() {
  if (!fs.existsSync(file)) { stateReadCache=null; return initial(); }
  const identity=stateFileIdentity(fs.statSync(file,{bigint:true}));
  if(stateReadCache?.identity===identity)return JSON.parse(stateReadCache.serialized);
  // Bind the cache to the file actually read, including external atomic replacements.
  const descriptor=fs.openSync(file,'r');let raw,readIdentity;
  try{const before=stateFileIdentity(fs.fstatSync(descriptor,{bigint:true}));raw=fs.readFileSync(descriptor,'utf8');const after=stateFileIdentity(fs.fstatSync(descriptor,{bigint:true}));readIdentity=before===after?before:null;}
  finally{fs.closeSync(descriptor);}
  const stored = JSON.parse(raw);
  if (!isEncryptedEnvelope(stored)) {
    if (host === '0.0.0.0') throw Error('plaintext_state_not_allowed_on_network_bind');
    stateReadCache=readIdentity?{identity:readIdentity,serialized:raw}:null;return stored;
  }
  if (!encryptionKey) throw Error('encrypted_state_key_missing');
  const serialized=decodeStateEnvelope(stored,encryptionKey);
  const state=JSON.parse(serialized);stateReadCache=readIdentity?{identity:readIdentity,serialized}:null;return state;
}
function save(state) {
  stateReadCache=null;
  conversationContexts.delete(state);
  reconcileVisitFeedback(state,candidates().rows,instance,{project:visitFeedback,isCurrent:(c,d)=>currentAppointmentAttempt(state,c,d)});
  state.presentation_history=presentationHistory(state,candidates().rows);
  fs.mkdirSync(path.dirname(file), { recursive: true, mode: 0o700 });
  const temp = `${file}.${process.pid}.tmp`;
  const serialized = JSON.stringify(state);
  let content = serialized;
  if (encryptionKey) {
    content = JSON.stringify(encodeStateEnvelope(content,encryptionKey));
  }
  fs.writeFileSync(temp, content, { mode: 0o600 });
  fs.renameSync(temp, file);
  // Bind the reusable plaintext to the bytes actually read after the save.
  // A concurrent external replacement/write must invalidate this optimization.
  try {
    const descriptor=fs.openSync(file,'r');
    try {
      const before=stateFileIdentity(fs.fstatSync(descriptor,{bigint:true}));
      const raw=fs.readFileSync(descriptor,'utf8');
      const after=stateFileIdentity(fs.fstatSync(descriptor,{bigint:true}));
      if(before===after&&raw===content)stateReadCache={identity:after,serialized};
    } finally { fs.closeSync(descriptor); }
  } catch { stateReadCache=null; } // The committed save must not fail for a cache miss.
}
